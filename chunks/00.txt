'use strict';

var AZ = window.AZ = window.AZ || {};
AZ.TILE = 32;
AZ.CHUNK_TILES = 24;
AZ.CHUNK_SIZE = AZ.TILE * AZ.CHUNK_TILES;

AZ.hash32 = function(seed,x,y,salt=0){
  let h=(seed ^ Math.imul(x,374761393) ^ Math.imul(y,668265263) ^ Math.imul(salt,2246822519))>>>0;
  h=Math.imul(h^(h>>>13),1274126177)>>>0; h^=h>>>16; return h>>>0;
};
AZ.rand01 = function(seed,x,y,salt=0){ return AZ.hash32(seed,x,y,salt)/4294967295; };
AZ.clamp=(v,a,b)=>Math.max(a,Math.min(b,v));
AZ.lerp=(a,b,t)=>a+(b-a)*t;
AZ.dist2=(a,b)=>{const x=a.x-b.x,y=a.y-b.y;return x*x+y*y};
AZ.chunkCoord=v=>Math.floor(v/AZ.CHUNK_SIZE);
AZ.localCoord=v=>((v%AZ.CHUNK_SIZE)+AZ.CHUNK_SIZE)%AZ.CHUNK_SIZE;

class World {
  constructor(seed){
    this.seed=seed>>>0;
    this.chunks=new Map();
    this.touched=new Map();
    this.maxRenderCache=96;
    this.falloutOrigin={x:AZ.CHUNK_SIZE*1.15,y:-AZ.CHUNK_SIZE*.4};
    this.wind={x:.72,y:.24};
    this.worldMinute=420;
  }
  key(cx,cy){return `${cx},${cy}`}
  biomeAtChunk(cx,cy){
    const n=AZ.rand01(this.seed,Math.floor(cx/2),Math.floor(cy/2),11);
    const wet=AZ.rand01(this.seed,Math.floor(cx/3),Math.floor(cy/3),19);
    if(wet<.09)return 'marsh';
    if(n<.28)return 'forest';
    if(n>.82)return 'farmland';
    return 'suburb';
  }
  roadMask(cx,cy){
    // Coherent infinite road lattice with local procedural branches.
    const vertical=(Math.abs(cx)%4===0) || (AZ.rand01(this.seed,cx,0,31)<.11);
    const horizontal=(Math.abs(cy)%4===0) || (AZ.rand01(this.seed,0,cy,37)<.11);
    return {vertical,horizontal};
  }
  getChunk(cx,cy){
    const k=this.key(cx,cy);
    let c=this.chunks.get(k); if(c){c.lastUsed=performance.now();return c;}
    c=this.touched.get(k); if(c){c.lastUsed=performance.now();this.chunks.set(k,c);this.touched.delete(k);return c;}
    c=this.generateChunk(cx,cy); this.chunks.set(k,c); this.prune(); return c;
  }
  generateChunk(cx,cy){
    const biome=this.biomeAtChunk(cx,cy), roads=this.roadMask(cx,cy), sites=[];
    const baseX=cx*AZ.CHUNK_SIZE,baseY=cy*AZ.CHUNK_SIZE;
    const count=biome==='suburb'?7:biome==='farmland'?4:biome==='forest'?2:2;
    const types=biome==='suburb'?['house','house','house','store','clinic','garage','warehouse']:biome==='farmland'?['farm','barn','house','shed']:['cabin','shed','tower'];
    for(let i=0;i<count;i++){
      const r1=AZ.rand01(this.seed,cx,cy,100+i*4),r2=AZ.rand01(this.seed,cx,cy,101+i*4);
      let lx=72+r1*(AZ.CHUNK_SIZE-144),ly=72+r2*(AZ.CHUNK_SIZE-144);
      if(roads.vertical && Math.abs(lx-AZ.CHUNK_SIZE/2)<105)lx+=lx<AZ.CHUNK_SIZE/2?-115:115;
      if(roads.horizontal && Math.abs(ly-AZ.CHUNK_SIZE/2)<105)ly+=ly<AZ.CHUNK_SIZE/2?-115:115;
      const type=types[Math.floor(AZ.rand01(this.seed,cx,cy,102+i*4)*types.length)];
      const w=type==='warehouse'?128:type==='farm'?112:80, h=type==='warehouse'?96:64;
      const loot=this.defaultLoot(type,cx,cy,i);
      sites.push({id:`${cx}:${cy}:${i}`,type,x:baseX+lx-w/2,y:baseY+ly-h/2,w,h,loot,integrity:100,barricade:0,claimedBy:null,fire:0,dirty:false});
    }
    const hotspot=AZ.rand01(this.seed,cx,cy,777)<.028;
    if(hotspot)sites.push({id:`${cx}:${cy}:rad`,type:'waste',x:baseX+AZ.CHUNK_SIZE*.68,y:baseY+AZ.CHUNK_SIZE*.32,w:72,h:72,loot:[{id:'iodine',name:'Iodeto de potássio',kind:'medical',qty:1}],integrity:65,barricade:0,claimedBy:null,fire:0,dirty:false});
    return {cx,cy,biome,roads,sites,hotspot,lastUsed:performance.now()};
  }
  defaultLoot(type,cx,cy,i){
    const q=(salt,min,max)=>min+Math.floor(AZ.rand01(this.seed,cx,cy,salt+i*13)*(max-min+1));
    const a=[];
    const push=(id,name,kind,qty,weight=1)=>{if(qty>0)a.push({id,name,kind,qty,weight});};
    if(['house','cabin','farm'].includes(type)){push('water','Água', 'drink',q(201,1,5),1);push('food','Alimento','food',q(202,1,5),.6);push('cloth','Tecido','material',q(203,0,3),.3)}
    if(type==='store'){push('water','Água','drink',q(204,5,14));push('food','Alimento','food',q(205,6,18),.6);push('battery','Bateria','utility',q(206,1,5),.2)}
    if(type==='clinic'){push('bandage','Curativo','medical',q(207,3,8),.2);push('antirad','Quelante','medical',q(208,1,4),.2);push('painkiller','Analgésico','medical',q(209,1,5),.1)}
    if(['garage','shed','barn'].includes(type)){push('wood','Madeira','material',q(210,2,8),1.5);push('metal','Metal','material',q(211,1,6),1.8);push('toolkit','Ferramentas','tool',q(212,0,2),2)}
    if(type==='warehouse'){push('wood','Madeira','material',q(213,5,14),1.5);push('metal','Metal','material',q(214,4,12),1.8);push('food','Alimento','food',q(215,1,8),.6)}
    return a;
  }
  radiationAt(x,y,minute=this.worldMinute){
    const dx=x-this.falloutOrigin.x,dy=y-this.falloutOrigin.y;
    const along=dx*this.wind.x+dy*this.wind.y;
    const cross=Math.abs(dx*this.wind.y-dy*this.wind.x);
    const plume=along>-250?Math.exp(-cross/900)*Math.exp(-Math.max(0,along)/8200):0;
    const age=Math.max(0,(minute-420)/1440);
    const decay=Math.exp(-age*.045);
    const cx=AZ.chunkCoord(x),cy=AZ.chunkCoord(y); const c=this.getChunk(cx,cy);
    let local=c.hotspot?Math.max(0,1-Math.hypot(AZ.localCoord(x)-AZ.CHUNK_SIZE*.68,AZ.localCoord(y)-AZ.CHUNK_SIZE*.32)/260):0;
    return AZ.clamp((plume*.9+local*.75)*decay,0,1.5);
  }
  terrainAt(x,y){
    const cx=AZ.chunkCoord(x),cy=AZ.chunkCoord(y),c=this.getChunk(cx,cy),lx=AZ.localCoord(x),ly=AZ.localCoord(y);
    if(c.roads.vertical && Math.abs(lx-AZ.CHUNK_SIZE/2)<46)return 'road';
    if(c.roads.horizontal && Math.abs(ly-AZ.CHUNK_SIZE/2)<46)return 'road';
    return c.biome;
  }
  nearbySites(x,y,radius=500){
    const out=[],r2=radius*radius,ccx=AZ.chunkCoord(x),ccy=AZ.chunkCoord(y),cr=Math.ceil(radius/AZ.CHUNK_SIZE)+1;
    for(let cy=ccy-cr;cy<=ccy+cr;cy++)for(let cx=ccx-cr;cx<=ccx+cr;cx++){
      const c=this.getChunk(cx,cy); for(const s of c.sites){const px=s.x+s.w/2,py=s.y+s.h/2;if((px-x)**2+(py-y)**2<=r2)out.push(s)}
    } return out;
  }
