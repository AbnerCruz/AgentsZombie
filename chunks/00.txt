(() => {

class RNG {
  constructor(seed=1){ this.s = seed >>> 0 || 1; }
  next(){ let x=this.s; x^=x<<13; x^=x>>>17; x^=x<<5; this.s=x>>>0; return this.s/4294967296; }
  int(a,b){ return Math.floor(this.next()*(b-a+1))+a; }
  pick(a){ return a[Math.floor(this.next()*a.length)]; }
  chance(p){ return this.next()<p; }
  range(a,b){ return a+(b-a)*this.next(); }
}

class Sim {

  idx(x,y){ return y*this.W+x; }
  inBounds(x,y){ return x>=0&&y>=0&&x<this.W&&y<this.H; }
  tileAt(x,y){ const tx=Math.floor(x/this.tile),ty=Math.floor(y/this.tile); return this.tiles[this.idx(tx,ty)]; }
  radAt(x,y){ const tx=clamp(Math.floor(x/this.tile),0,this.W-1),ty=clamp(Math.floor(y/this.tile),0,this.H-1); return this.rad[this.idx(tx,ty)]; }

  generateWorld(){
    for(let y=0;y<this.H;y++) for(let x=0;x<this.W;x++){
      const edge=Math.min(x,y,this.W-1-x,this.H-1-y);
      this.tiles.push({type: edge<3?'forest':'grass', variant:this.rng.int(0,3)});
    }
    const setTile=(x,y,type)=>{ if(this.inBounds(x,y)) this.tiles[this.idx(x,y)]={type,variant:this.rng.int(0,3)}; };
    // Roads: irregular but coherent small-town grid.
    const hRoads=[18,35,53,69], vRoads=[22,45,67,88,104];
    hRoads.forEach((ry,i)=>{
      for(let x=4;x<this.W-4;x++) for(let dy=-1;dy<=1;dy++) setTile(x,ry+dy,'road');
      this.roads.push({a:{x:4,y:ry},b:{x:this.W-4,y:ry}});
    });
    vRoads.forEach(rx=>{
      for(let y=5;y<this.H-5;y++) for(let dx=-1;dx<=1;dx++) setTile(rx+dx,y,'road');
      this.roads.push({a:{x:rx,y:5},b:{x:rx,y:this.H-5}});
    });
    // River and contaminated drainage east side.
    for(let y=0;y<this.H;y++){
      const cx=112+Math.sin(y*.16)*2;
      for(let x=Math.floor(cx)-1;x<=Math.floor(cx)+1;x++) setTile(x,y,'water');
    }
    // Forest belts.
    for(let y=3;y<this.H-3;y++) for(let x=3;x<this.W-3;x++){
      if(this.tiles[this.idx(x,y)].type==='grass' && (x<13 || y>76 || (x>94&&y<16)) && this.rng.chance(.42)) setTile(x,y,'forest');
    }

    const addBuilding=(name,x,y,w,h,type,capacity=12)=>{
      const b={id:'b'+this.buildings.length,name,x:x*this.tile,y:y*this.tile,w:w*this.tile,h:h*this.tile,type,capacity,food:0,water:0,meds:0,fort:0,power:true,occupants:new Set()};
      if(type==='grocery'){b.food=120;b.water=90} if(type==='hospital'){b.meds=70;b.water=40} if(type==='farm'){b.food=80;b.water=70} if(type==='home'){b.food=this.rng.int(4,18);b.water=this.rng.int(5,20)} if(type==='school'){b.food=25;b.water=40} if(type==='police'){b.food=16;b.water=25;b.fort=.35} if(type==='radio'){b.food=8;b.water=15;b.fort=.2}
      this.buildings.push(b); return b;
    };
    // Homes across neighborhoods.
    const homeSpots=[];
    [[8,8,18,16],[27,8,16,17],[49,8,15,17],[72,8,13,17],[8,40,12,11],[28,40,13,11],[72,40,12,11],[92,40,11,11],[8,58,12,10],[28,58,12,10],[72,58,12,10],[92,58,11,10]].forEach(([sx,sy,w,h])=>{
      for(let yy=sy;yy<sy+h;yy+=6) for(let xx=sx;xx<sx+w;xx+=7) if(this.rng.chance(.82)) homeSpots.push([xx,yy]);
    });
    homeSpots.slice(0,34).forEach((p,i)=>addBuilding('Casa '+(i+1),p[0],p[1],4,3,'home',6));
    // Civic / commercial / survival points.
    this.poi.push(addBuilding('Hospital Municipal',48,27,10,6,'hospital',35));
    this.poi.push(addBuilding('Mercado Ashfall',71,27,9,6,'grocery',24));
    this.poi.push(addBuilding('Escola Comunitária',27,27,10,6,'school',45));
    this.poi.push(addBuilding('Delegacia',86,27,7,6,'police',18));
    this.poi.push(addBuilding('Estação KACR 91.7',95,61,7,5,'radio',14));
    this.poi.push(addBuilding('Corpo de Bombeiros',14,27,7,6,'fire',18));
    this.poi.push(addBuilding('Oficina Oakes',48,59,8,5,'shop',12));
    this.poi.push(addBuilding('Farmácia Central',60,27,6,5,'pharmacy',10));
    this.poi.push(addBuilding('Fazenda Bell',14,72,12,7,'farm',16));
    this.poi.push(addBuilding('Abrigo Municipal',81,72,10,6,'shelter',60));
    this.poi.push(addBuilding('Complexo Nuclear Cinder Lake',106,5,10,9,'plant',25));
    this.poi.push(addBuilding('Posto de Gasolina',60,59,7,5,'gas',12));

    // Mark building floor footprints.
    for(const b of this.buildings){
      const sx=Math.floor(b.x/this.tile), sy=Math.floor(b.y/this.tile), w=Math.floor(b.w/this.tile),h=Math.floor(b.h/this.tile);
      for(let y=sy;y<sy+h;y++) for(let x=sx;x<sx+w;x++) if(this.inBounds(x,y)) setTile(x,y,'lot');
    }
  }

  createPopulation(agentCount,npcCount){
    const n=agentCount+npcCount;
    const homes=this.buildings.filter(b=>b.type==='home');
    const workplaces=this.buildings.filter(b=>['hospital','grocery','school','police','radio','fire','shop','pharmacy','farm','gas'].includes(b.type));
    const householdLast=[];
    for(let i=0;i<homes.length;i++) householdLast[i]=this.rng.pick(LAST);
    for(let i=0;i<n;i++){
      const hIndex=i%homes.length;
      const home=homes[hIndex];
      const age=this.rng.int(8,84);
      const job=age>67?'aposentado':age<18?'estudante':this.rng.pick(JOBS);
      const work=this.findWorkplaceForJob(job,workplaces);
      const cognitive=i<agentCount?'agent':'npc'; // Internal simulation detail; never exposed to inhabitants.
      const a={
        kind:'human',id:i+1,name:`${this.rng.pick(FIRST)} ${householdLast[hIndex]}`,age,job,trait:this.rng.pick(TRAITS),cognitive,
        x:home.x+this.rng.range(8,home.w-8),y:home.y+this.rng.range(8,home.h-8),vx:0,vy:0,
        home,work,health:100,hunger:this.rng.range(3,20),thirst:this.rng.range(2,18),fatigue:this.rng.range(2,22),stress:this.rng.range(4,18),radiation:0,infection:0,
        courage:this.rng.range(.25,.85),empathy:this.rng.range(.25,.9),sociability:this.rng.range(.2,.95),awareness:this.rng.range(.3,.85),
        leadership:this.rng.range(.15,.92),risk:this.rng.range(.15,.9),
        food:this.rng.int(0,2),water:this.rng.int(0,2),meds:0,weapon:this.rng.chance(.12),hasRadio:this.rng.chance(.38),alive:true,turned:false,
        action:'home',target:{x:home.x+home.w/2,y:home.y+home.h/2},group:null,memories:[],path:[],decisionIn:this.rng.range(0,.8),panic:false,knownDanger:0,radioKnowledge:0,lastBuilding:home,
        family:[],kills:0,helped:0,beliefs:new Set(),rumors:[],relations:new Map(),socialCooldown:this.rng.range(2,12),goal:null
      };
      this.agents.push(a);
    }
    // Household relationships are real to everyone; nobody has access to simulation tiers.
    for(const a of this.agents){
      a.family=this.agents.filter(b=>b!==a&&b.home===a.home).map(b=>b.id);
      for(const id of a.family)a.relations.set(id,this.rng.range(.58,.92));
    }
    this.seedMorningRoutine();
  }

  seedMorningRoutine(){
    for(const a of this.agents){
      const roll=this.rng.next();
      if(a.work && roll<.38){
        a.x=a.work.x+this.rng.range(8,Math.max(9,a.work.w-8));
        a.y=a.work.y+this.rng.range(8,Math.max(9,a.work.h-8));
        a.lastBuilding=a.work;
        a.action='wander';
      }else if(a.work && roll<.72){
        const t=this.rng.range(.15,.85);
        a.x=lerp(a.home.x+a.home.w/2,a.work.x+a.work.w/2,t)+this.rng.range(-20,20);
        a.y=lerp(a.home.y+a.home.h/2,a.work.y+a.work.h/2,t)+this.rng.range(-20,20);
        a.target={x:a.work.x+a.work.w/2,y:a.work.y+a.work.h/2};
        a.action='wander';
      }
      if(a.job==='policial'||a.job==='bombeiro'||a.job==='enfermeira'||a.job==='médico')a.beliefs.add('emergency-traffic');
      if(a.job==='radialista')a.beliefs.add('radio-active');
      if(a.job==='técnica de laboratório'&&this.rng.chance(.55))a.beliefs.add('cinder-anomaly');
    }
  }

  findWorkplaceForJob(job, workplaces){
    const map={enfermeira:'hospital',médico:'hospital',professor:'school',policial:'police',bombeiro:'fire',radialista:'radio',mecânico:'shop',caixa:'grocery',farmacêutica:'pharmacy',fazendeiro:'farm',motorista:'gas',cozinheira:'grocery',segurança:'police','técnica de laboratório':'hospital',carpinteiro:'shop',estudante:'school'};
    const t=map[job];
    return workplaces.find(b=>b.type===t) || this.rng.pick(workplaces);
  }

  seedCrisis(){
    // Every run begins at Day 1, 07:00. Only conditions are seeded; outcomes are not scripted.
    const plant=this.buildings.find(b=>b.type==='plant');
    const px=Math.floor((plant.x+plant.w/2)/this.tile),py=Math.floor((plant.y+plant.h/2)/this.tile);
    for(let y=0;y<this.H;y++) for(let x=0;x<this.W;x++){
      const d=Math.hypot(x-px,y-py);
      this.rad[this.idx(x,y)]=clamp((17-d)/17,0,1)*.55;
    }
    // Seven residents are already incubating. Their neighbors only know what they can observe or hear.
    const shuffled=[...this.agents].sort(()=>this.rng.next()-.5);
    shuffled.slice(0,7).forEach((a,i)=>{a.infection=i<2?this.rng.range(.78,.92):this.rng.range(.32,.72);a.stress+=this.rng.range(10,25);this.remember(a,'Acordou com febre e sintomas incomuns.');});
    shuffled.slice(7,23).forEach(a=>a.beliefs.add('violent-incidents'));
    shuffled.slice(23,34).forEach(a=>a.beliefs.add('cinder-anomaly'));
    shuffled.slice(34,52).forEach(a=>a.beliefs.add('radio-active'));

    // A handful of already-turned people ignite the first visible panic without defining the future.
    const outbreakPoints=[{x:76*this.tile,y:31*this.tile},{x:53*this.tile,y:31*this.tile},{x:89*this.tile,y:31*this.tile},{x:64*this.tile,y:55*this.tile},{x:33*this.tile,y:32*this.tile},{x:21*this.tile,y:28*this.tile}];
    outbreakPoints.forEach((p,i)=>this.zombies.push({kind:'zombie',source:null,id:`outbreak-${i+1}`,name:'Pessoa não identificada',x:p.x+this.rng.range(-18,18),y:p.y+this.rng.range(-18,18),target:this.randomRoadPoint(),alive:true,health:this.rng.range(58,84),mutation:this.rng.range(0,.08),decay:0}));

    this.log('07:00','EMERGÊNCIA','Sirene civil interrompe a manhã. A rede móvel começa a ficar instável.',true);
    this.log('07:03','RÁDIO','KACR relata incidentes violentos sem causa confirmada no condado.',true);
    this.log('07:07','CINDER LAKE','Sensores registram liberação anormal no complexo nuclear a leste.',true);
  }

  bindUI(){
    $('pauseBtn').onclick=()=>{this.paused=!this.paused;$('pauseBtn').textContent=this.paused?'▶':'Ⅱ';};
    document.querySelectorAll('.speed').forEach(btn=>btn.onclick=()=>{this.speed=+btn.dataset.speed;document.querySelectorAll('.speed').forEach(b=>b.classList.toggle('active',b===btn));});
    $('resetBtn').onclick=()=>{ if(confirm('Iniciar uma nova simulação no Dia 1?')) location.reload(); };
    $('radToggle').onchange=e=>this.overlayRad=e.target.checked;
    $('territoryToggle').onchange=e=>this.overlayTerritory=e.target.checked;
    $('pathToggle').onchange=e=>this.overlayPath=e.target.checked;
    $('clearFollow').onclick=()=>{this.follow=null;};
    $('searchAgent').oninput=e=>this.search(e.target.value);

    canvas.addEventListener('pointerdown',e=>{
      canvas.setPointerCapture(e.pointerId); this.pointer.drag=true; this.pointer.moved=false; this.pointer.x=this.pointer.lastX=e.clientX;this.pointer.y=this.pointer.lastY=e.clientY;
    });
    canvas.addEventListener('pointermove',e=>{
      if(!this.pointer.drag)return; const dx=e.clientX-this.pointer.lastX,dy=e.clientY-this.pointer.lastY;
      if(Math.abs(e.clientX-this.pointer.x)+Math.abs(e.clientY-this.pointer.y)>6)this.pointer.moved=true;
      if(!this.follow){this.camera.x-=dx/this.camera.zoom;this.camera.y-=dy/this.camera.zoom;}
      this.pointer.lastX=e.clientX;this.pointer.lastY=e.clientY;
    });
    canvas.addEventListener('pointerup',e=>{if(!this.pointer.moved)this.selectAt(e.clientX,e.clientY);this.pointer.drag=false;});
    canvas.addEventListener('wheel',e=>{e.preventDefault();const before=this.screenToWorld(e.clientX,e.clientY);this.camera.zoom=clamp(this.camera.zoom*(e.deltaY>0?.88:1.12),.32,2.7);const after=this.screenToWorld(e.clientX,e.clientY);this.camera.x+=before.x-after.x;this.camera.y+=before.y-after.y;},{passive:false});
    canvas.addEventListener('touchstart',e=>{if(e.touches.length===2){const [a,b]=e.touches;this.pinch={d:Math.hypot(a.clientX-b.clientX,a.clientY-b.clientY),z:this.camera.zoom};}},{passive:true});
    canvas.addEventListener('touchmove',e=>{if(e.touches.length===2&&this.pinch){const [a,b]=e.touches;const d=Math.hypot(a.clientX-b.clientX,a.clientY-b.clientY);this.camera.zoom=clamp(this.pinch.z*d/this.pinch.d,.32,2.7);}},{passive:true});
    canvas.addEventListener('touchend',()=>this.pinch=null,{passive:true});
  }

  search(q){
    q=q.trim().toLowerCase(); const box=$('searchResults'); box.innerHTML=''; if(!q)return;
    this.agents.filter(a=>a.name.toLowerCase().includes(q)).slice(0,8).forEach(a=>{const d=document.createElement('div');d.className='searchItem';d.textContent=`${a.name} · ${a.alive?(a.turned?'zumbi':'vivo'):'morto'}`;d.onclick=()=>{const target=a.turned?(this.zombies.find(z=>z.source===a&&z.alive)||a):a;this.selected=target;this.follow=target;this.renderInspector();box.innerHTML='';$('searchAgent').value='';};box.appendChild(d);});
  }

  resize(){ const dpr=Math.min(devicePixelRatio||1,2); canvas.width=Math.floor(innerWidth*dpr);canvas.height=Math.floor(innerHeight*dpr);canvas.style.width=innerWidth+'px';canvas.style.height=innerHeight+'px';this.dpr=dpr; }

  frame(now){
    const dt=Math.min(.05,(now-this.last)/1000); this.last=now;
    if(!this.paused){
      this.simAccumulator += dt*this.speed;
      let loops=0;
      while(this.simAccumulator>=.20 && loops<80){this.step(.20);this.simAccumulator-=.20;loops++;}
    }
    if(this.follow && this.follow.alive!==false){this.camera.x=lerp(this.camera.x,this.follow.x,.08);this.camera.y=lerp(this.camera.y,this.follow.y,.08);}
    this.draw(); this.refreshHUD();
    requestAnimationFrame(t=>this.frame(t));
  }

  step(dt){
    const minutes=dt*1.35; // 1 real simulation second ~= 6.75 game minutes at 1x.
    this.time+=minutes;
    while(this.time>=1440){this.time-=1440;this.day++;this.log(this.timeText(),'NOVO DIA',`O Dia ${this.day} começa com ${this.agents.filter(a=>a.alive&&!a.turned).length} sobreviventes.`,true);}
    this.updateWorld(dt,minutes);
    this.rebuildSpatial();
    for(const a of this.agents) if(a.alive&&!a.turned) this.updateHuman(a,dt,minutes);
    for(const z of this.zombies) if(z.alive) this.updateZombie(z,dt,minutes);
    this.updateGroups(dt);
    this.updateCorpses(minutes);
  }

  updateWorld(dt,minutes){
    const totalMinutes=(this.day-1)*1440+this.time;
    // Chaos is an emergent aggregate but the first day starts with failing institutions.
    const zombies=this.zombies.filter(z=>z.alive).length;
    const infected=this.agents.filter(a=>a.alive&&!a.turned&&a.infection>.05).length;
    const deaths=this.agents.filter(a=>!a.alive).length;
    const pop=Math.max(1,this.agents.length);const pressure=clamp(zombies/(pop*.20)+infected/(pop*.27)+deaths/(pop*.22),0,1);
    const dayOneRamp=clamp((totalMinutes-420)/540,0,.52);
    this.chaos=clamp(this.chaos + (pressure*.004 + dayOneRamp*.0015)*dt - .0004*dt,0,1);
    this.infrastructure=clamp(this.infrastructure-(this.chaos*.00042+zombies*.0000025)*minutes,0,1);

    // Radiation plume diffuses slowly with wind. A damaged plant worsens during first day.
    const plant=this.buildings.find(b=>b.type==='plant');
    const px=Math.floor((plant.x+plant.w/2)/this.tile),py=Math.floor((plant.y+plant.h/2)/this.tile);
    const release = totalMinutes<1100 ? .00065 : .00022;
    for(let k=0;k<6;k++){
      const x=clamp(px+this.rng.int(-3,3),0,this.W-1), y=clamp(py+this.rng.int(-3,3),0,this.H-1);
      this.rad[this.idx(x,y)]=clamp(this.rad[this.idx(x,y)]+release*minutes*6,0,1);
    }
    if(this.rng.chance(.03*dt)){
      const old=this.rad.slice();
      for(let y=1;y<this.H-1;y++) for(let x=1;x<this.W-1;x++){
        const i=this.idx(x,y); const upwindX=clamp(Math.round(x-this.wind.x),0,this.W-1),upwindY=clamp(Math.round(y-this.wind.y),0,this.H-1);
        const avg=(old[i]*5+old[this.idx(x-1,y)]+old[this.idx(x+1,y)]+old[this.idx(x,y-1)]+old[this.idx(x,y+1)]+old[this.idx(upwindX,upwindY)]*2)/11;
        this.rad[i]=clamp(avg*.9998,0,1);
      }
    }

    // Organic fire outbreaks become more likely as infrastructure collapses.
    if(this.rng.chance((.0003+this.chaos*.0018)*dt)){
      const x=this.rng.int(8,this.W-9),y=this.rng.int(8,this.H-9); if(this.tiles[this.idx(x,y)].type!=='water'){this.fire[this.idx(x,y)]=1;this.log(this.timeText(),'INCÊNDIO',`Fogo avistado no setor ${String.fromCharCode(65+Math.floor(x/20))}-${Math.floor(y/15)+1}.`);}
    }

  updateHuman(a,dt,minutes){

    if(a.hunger>82)a.health-=minutes*.018;if(a.thirst>78)a.health-=minutes*.032;
    if(a.infection>0){a.infection=clamp(a.infection+minutes*(.00062+.00034*(a.health<55)),0,1.2);a.health-=minutes*a.infection*.006;a.stress=clamp(a.stress+minutes*a.infection*.01,0,100);}
    if(a.infection>=1){this.turn(a);return;} if(a.health<=0){this.killHuman(a,'ferimentos e colapso físico');return;}

    // Perception.
    const dangerZ=this.nearestZombie(a,95);
    if(dangerZ){a.knownDanger=clamp(a.knownDanger+.25,0,1);a.stress=clamp(a.stress+minutes*.12,0,100); if(dist(a,dangerZ)<17){this.zombieAttack(dangerZ,a,dt);}}
    else a.knownDanger=Math.max(0,a.knownDanger-dt*.01);
    const nearbyHuman=this.nearestHuman(a,48);
    if(nearbyHuman && a.stress>55 && a.sociability>.45){a.stress=Math.max(0,a.stress-minutes*.018);}
    a.socialCooldown-=dt;
    if(nearbyHuman && !dangerZ && a.socialCooldown<=0){this.socialExchange(a,nearbyHuman);a.socialCooldown=this.rng.range(8,24);}

    a.decisionIn-=dt;
    if(a.decisionIn<=0){this.decide(a);a.decisionIn=a.cognitive==='agent'?this.rng.range(.8,2.8):this.rng.range(2.2,5.5);}
    this.moveHuman(a,dt,dangerZ);

  }

  decide(a){
    const scores={};
    const rad=this.radAt(a.x,a.y), danger=this.nearestZombie(a,82);
    scores.escapeRadiation=rad*130 + (a.radiation>45?30:0);
    scores.flee=danger ? 95-clamp(dist(a,danger),0,95)+a.stress*.25 : 0;
    scores.water=a.thirst*.95 + (a.water===0?12:0);
    scores.food=a.hunger*.78 + (a.food===0?8:0);
    scores.medical=(100-a.health)*.65 + a.infection*42 + (a.radiation>45?22:0);
    scores.rest=a.fatigue*.56;
    const missingFamily=a.family.map(id=>this.personById(id)).filter(f=>f&&f.alive&&!f.turned&&dist(a,f)>90);
    scores.group=missingFamily.length ? 22+a.sociability*28+a.stress*.22 : a.sociability*8;
    const injured=this.nearestHuman(a,75,h=>h.health<55||h.infection>.35);
    scores.help=injured ? a.empathy*55+(a.job==='médico'||a.job==='enfermeira'?30:0)-a.stress*.25 : 0;
    scores.shelter=this.chaos*38+a.stress*.25+(this.infrastructure<.5?12:0);
    scores.scavenge=(a.food+a.water<3?26:8)+this.chaos*12;
    scores.home=(a.family.length?18:7)+a.stress*.16;
    scores.radio=(a.radioKnowledge<.5&&a.sociability>.45)?12+this.chaos*20:0;
    scores.wander=6;
    if(a.trait==='protetor')scores.help+=12;if(a.trait==='corajoso')scores.flee-=8;if(a.trait==='ansioso')scores.flee+=14;if(a.trait==='pragmático'){scores.food+=5;scores.water+=5;scores.shelter+=5}if(a.trait==='solitário')scores.group-=14;if(a.trait==='altruísta')scores.help+=15;
    if(a.beliefs.has('shelter-open'))scores.shelter+=11;
    if(a.beliefs.has('hospital-overflow'))scores.medical-=8;
    if(a.beliefs.has('east-radiation')||a.beliefs.has('cinder-anomaly'))scores.escapeRadiation+=rad*18;
    if(a.group&&a.group.leaderId!==a.id&&a.group.base)scores.shelter+=a.group.cohesion*14;
    // Full cognitive agents maintain stronger long-horizon social motives; NPCs still obey the same world rules and are indistinguishable to others.
    if(a.cognitive==='agent'){
      if(a.family.some(id=>{const f=this.personById(id);return f&&f.alive&&!f.turned&&dist(a,f)>130;}))scores.group+=14;
      if(a.group&&a.group.leaderId===a.id)scores.shelter+=8+a.leadership*8;
      scores.help+=a.empathy*5;
    }
    // Noise makes behavior human rather than perfectly rational.
    Object.keys(scores).forEach(k=>scores[k]+=this.rng.range(-5,5));
    let action=Object.entries(scores).sort((x,y)=>y[1]-x[1])[0][0];
    a.action=action;
    this.setTargetForAction(a,action,{danger,injured,missingFamily});
  }

  setTargetForAction(a,action,ctx){
    const center=b=>({x:b.x+b.w/2,y:b.y+b.h/2});
    if(action==='flee' && ctx.danger){const dx=a.x-ctx.danger.x,dy=a.y-ctx.danger.y,l=Math.hypot(dx,dy)||1;a.target={x:clamp(a.x+dx/l*140,12,this.W*this.tile-12),y:clamp(a.y+dy/l*140,12,this.H*this.tile-12)};return;}
    if(action==='escapeRadiation'){a.target=this.findLowerRadiationPoint(a);return;}
    if(action==='water'){const b=this.bestBuilding(a,b=>b.water>0||['farm','hospital','school','grocery'].includes(b.type));a.target=b?center(b):this.randomRoadPoint();return;}
    if(action==='food'){const b=this.bestBuilding(a,b=>b.food>0||['farm','grocery'].includes(b.type));a.target=b?center(b):this.randomRoadPoint();return;}
    if(action==='medical'){const b=this.buildings.find(b=>b.type==='hospital');a.target=center(b);return;}
    if(action==='rest'||action==='home'){a.target=center(a.home);return;}
    if(action==='group'&&ctx.missingFamily.length){const f=ctx.missingFamily.reduce((p,c)=>dist(a,c)<dist(a,p)?c:p);a.target={x:f.x,y:f.y};return;}
    if(action==='help'&&ctx.injured){a.target={x:ctx.injured.x,y:ctx.injured.y};return;}
    if(action==='radio'){a.target=center(this.buildings.find(b=>b.type==='radio'));return;}
    if(action==='shelter'){const choices=this.buildings.filter(b=>['shelter','school','police','hospital'].includes(b.type));const b=choices.reduce((p,c)=>(this.safetyOf(c)-dist(a,c)*.0008)>(this.safetyOf(p)-dist(a,p)*.0008)?c:p);a.target=center(b);return;}
    if(action==='scavenge'){const b=this.bestBuilding(a,b=>!['plant'].includes(b.type)&&dist(a,b)<500);a.target=b?center(b):this.randomRoadPoint();return;}
    a.target=this.randomRoadPoint();
  }

  safetyOf(b){return b.fort*.5 + (1-this.radAt(b.x+b.w/2,b.y+b.h/2))*.5;}
  bestBuilding(a,pred){const arr=this.buildings.filter(pred);if(!arr.length)return null;return arr.reduce((p,c)=>dist(a,c)<dist(a,p)?c:p);}
  randomRoadPoint(){const r=this.rng.pick(this.roads);const t=this.rng.next();return{x:lerp(r.a.x,r.b.x,t)*this.tile,y:lerp(r.a.y,r.b.y,t)*this.tile};}
  findLowerRadiationPoint(a){let best={x:a.x,y:a.y,r:this.radAt(a.x,a.y)};for(let i=0;i<18;i++){const ang=this.rng.next()*TAU,d=this.rng.range(80,280),x=clamp(a.x+Math.cos(ang)*d,12,this.W*this.tile-12),y=clamp(a.y+Math.sin(ang)*d,12,this.H*this.tile-12),r=this.radAt(x,y);if(r<best.r)best={x,y,r};}return best;}

  moveHuman(a,dt,danger){

    for(const b of this.nearHumans(a,14)){if(b===a)continue;const dd=dist(a,b);if(dd<11&&dd>0){sx+=(a.x-b.x)/dd*(11-dd);sy+=(a.y-b.y)/dd*(11-dd);}}

  tryFormGroup(a){

      const old=g.leaderId;g.leaderId=leader.id;

  updateGroups(dt){

      if(members.length<2)continue;
      if(!g.leaderId||!this.personById(g.leaderId)?.alive)this.electLeader(g);
      const avgStress=members.reduce((sum,a)=>sum+a.stress,0)/members.length;
      const avgTrust=members.reduce((sum,a)=>sum+this.relationship(a,this.personById(g.leaderId)||a),0)/members.length;
      g.cohesion=clamp(g.cohesion+(members.length*.00002+avgTrust*.00004-avgStress*.0000018)*dt,0,1);

        if(leaver){g.members.delete(leaver.id);leaver.group=null;this.remember(leaver,`Abandonou ${g.name} após conflitos.`);}

  updateZombie(z,dt,minutes){

    }

  }

  }
  nearHumans(a,r){return this.nearbyFromGrid(this.humanGrid,a,r).filter(h=>h.alive&&!h.turned&&dist(a,h)<=r);}
  nearZombies(a,r){return this.nearbyFromGrid(this.zombieGrid,a,r).filter(z=>z.alive&&dist(a,z)<=r);}
  nearestZombie(a,r){let best=null,bd=r;for(const z of this.nearZombies(a,r)){if(z===a)continue;const d=dist(a,z);if(d<bd){bd=d;best=z;}}return best;}
  nearestHuman(a,r,pred=()=>true){let best=null,bd=r;for(const h of this.nearHumans(a,r)){if(h===a||!pred(h))continue;const d=dist(a,h);if(d<bd){bd=d;best=h;}}return best;}
  personById(id){return this.agents[id-1]||null;}
  buildingAt(x,y){return this.buildings.find(b=>x>=b.x&&y>=b.y&&x<=b.x+b.w&&y<=b.y+b.h);}
  remember(a,text){if(!a.memories)return;a.memories.unshift(text);const max=a.cognitive==='agent'?12:4;if(a.memories.length>max)a.memories.length=max;}

  relationship(a,b){
    if(!a||!b||a===b)return .5;
    if(!a.relations.has(b.id))a.relations.set(b.id,a.family.includes(b.id)?this.rng.range(.58,.9):this.rng.range(-.08,.22));
    return a.relations.get(b.id);
  }
  adjustRelationship(a,b,delta){if(!a||!b||a===b)return;const v=clamp(this.relationship(a,b)+delta,-1,1);a.relations.set(b.id,v);}
  socialExchange(a,b){
    const baseTrust=(this.relationship(a,b)+this.relationship(b,a))/2;
    const kin=a.family.includes(b.id)||b.family.includes(a.id);
    const delta=(kin?.025:.008)+(a.stress>75&&b.stress>75?-.012:0);
    this.adjustRelationship(a,b,delta);this.adjustRelationship(b,a,delta*.8);
    if(a.beliefs.size&&this.rng.chance(.58+a.sociability*.22)){
      const belief=this.rng.pick([...a.beliefs]);b.beliefs.add(belief);
      if(b.cognitive==='agent'&&this.rng.chance(.22))this.remember(b,`${a.name} contou algo sobre ${this.beliefLabel(belief)}.`);
    }
    if(b.beliefs.size&&this.rng.chance(.52+b.sociability*.2))a.beliefs.add(this.rng.pick([...b.beliefs]));
    if(baseTrust>.45&&a.stress>45)b.stress=Math.max(0,b.stress-2);
  }
  beliefLabel(key){return ({'cinder-anomaly':'problemas em Cinder Lake','east-radiation':'radiação a leste','violent-incidents':'ataques violentos','radio-active':'a rádio KACR','shelter-open':'um abrigo em formação','hospital-overflow':'o hospital lotado','emergency-traffic':'chamados de emergência'})[key]||'o que está acontecendo';}

  updateRadio(totalMinutes){
    const slot=Math.floor(totalMinutes/45);if(slot===this.lastRadioSlot)return;this.lastRadioSlot=slot;
    const station=this.buildings.find(b=>b.type==='radio');if(!station||!station.power||this.infrastructure<.18)return;
    const staff=this.agents.filter(a=>a.alive&&!a.turned&&dist(a,{x:station.x+station.w/2,y:station.y+station.h/2})<90&&(a.job==='radialista'||a.sociability>.68));
    if(!staff.length&&slot>11)return;
    let belief='violent-incidents',text='KACR pede que moradores evitem deslocamentos desnecessários.';
    if(this.radAverage()>.08){belief='east-radiation';text='KACR transmite alerta incompleto sobre contaminação vinda do leste.';}
    if(this.infrastructure<.58){belief='shelter-open';text='KACR compartilha relatos de abrigos improvisados e falhas de infraestrutura.';}
    if(this.agents.filter(a=>a.alive&&!a.turned&&a.health<60).length>18){belief='hospital-overflow';text='KACR relata saturação do Hospital Municipal.';}
    this.log(this.timeText(),'KACR 91.7',text);
    for(const a of this.agents)if(a.alive&&!a.turned&&a.hasRadio&&this.rng.chance(.62)){a.beliefs.add(belief);a.radioKnowledge=clamp(a.radioKnowledge+.16,0,1);}
  }

  updateCorpses(minutes){
    for(const c of this.corpses){
      c.age+=minutes;
      if(c.infected&&!c.reanimated&&c.age>=c.reanimateAt){
        c.reanimated=true;
        const z={kind:'zombie',source:c.source,id:`corpse-${c.source.id}-${Math.floor(this.totalTime())}`,name:c.name,x:c.x,y:c.y,target:this.randomRoadPoint(),alive:true,health:this.rng.range(42,68),mutation:clamp((c.source.radiation||0)/150,0,.55),decay:.08};
        this.zombies.push(z);this.log(this.timeText(),'REANIMAÇÃO',`${c.name} volta a se mover após a morte.`,true);
      }
    }
  }

  totalTime(){return (this.day-1)*1440+this.time;}
  timeText(){const h=Math.floor(this.time/60)%24,m=Math.floor(this.time%60);return `${String(h).padStart(2,'0')}:${String(m).padStart(2,'0')}`;}
  radAverage(){let s=0;for(let i=0;i<this.rad.length;i++)s+=this.rad[i];return s/this.rad.length;}

  log(time,type,text,important=false){
    const e={id:this.nextEventId++,time,type,text,important};this.events.unshift(e);if(this.events.length>120)this.events.pop();
    const feed=$('eventFeed');if(feed){const row=document.createElement('div');row.className='event';row.innerHTML=`<time>${time}</time><span><strong>${type}</strong> · ${text}</span>`;feed.prepend(row);while(feed.children.length>40)feed.lastChild.remove();}
  }

  refreshHUD(){
    const living=this.agents.filter(a=>a.alive&&!a.turned).length, infected=this.agents.filter(a=>a.alive&&!a.turned&&a.infection>.05).length, dead=this.agents.filter(a=>!a.alive).length, zombies=this.zombies.filter(z=>z.alive).length;
    $('aliveStat').textContent=living;$('infectedStat').textContent=infected;$('deadStat').textContent=dead;$('zombieStat').textContent=zombies;
    $('clock').textContent=`DIA ${this.day} · ${this.timeText()}`;
    const rav=this.radAverage(); $('chaosText').textContent=Math.round(this.chaos*100)+'%';$('chaosBar').style.width=this.chaos*100+'%';$('radText').textContent=Math.round(rav*100)+'%';$('radBar').style.width=rav*100+'%';$('infraText').textContent=Math.round(this.infrastructure*100)+'%';$('infraBar').style.width=this.infrastructure*100+'%';
    if(this.selected && this._lastInspectorTick!==(Math.floor(performance.now()/600))){this._lastInspectorTick=Math.floor(performance.now()/600);this.renderInspector();}
  }

  renderInspector(){
    const box=$('inspector'),o=this.selected;if(!o){box.innerHTML='<div class="emptyState"><h2>OBSERVADOR</h2><p>Toque em uma pessoa, zumbi ou ponto da cidade para acompanhar.</p></div>';return;}

      const status=!o.alive?'Morto':o.turned?'Transformado':o.infection>.05?'Infectado':'Sobrevivente';

  selectAt(sx,sy){
    const p=this.screenToWorld(sx,sy);let best=null,bd=16/this.camera.zoom;
    for(const a of this.agents){if(!a.alive||a.turned)continue;const d=Math.hypot(a.x-p.x,a.y-p.y);if(d<bd){bd=d;best=a;}}
    for(const z of this.zombies){if(!z.alive)continue;const d=Math.hypot(z.x-p.x,z.y-p.y);if(d<bd){bd=d;best=z;}}
    if(!best){const b=this.buildingAt(p.x,p.y);if(b)best={kind:'building',data:b};}
    this.selected=best;this.renderInspector();
  }

  screenToWorld(sx,sy){const rect=canvas.getBoundingClientRect();const x=(sx-rect.left-rect.width/2)/this.camera.zoom+this.camera.x,y=(sy-rect.top-rect.height/2)/this.camera.zoom+this.camera.y;return{x,y};}

  draw(){
    const dpr=this.dpr||1,w=canvas.width,h=canvas.height;ctx.setTransform(dpr,0,0,dpr,0,0);ctx.fillStyle='#0b100d';ctx.fillRect(0,0,w/dpr,h/dpr);
    ctx.save();ctx.translate(innerWidth/2,innerHeight/2);ctx.scale(this.camera.zoom,this.camera.zoom);ctx.translate(-this.camera.x,-this.camera.y);
    const vw=innerWidth/this.camera.zoom,vh=innerHeight/this.camera.zoom;const minX=clamp(Math.floor((this.camera.x-vw/2)/this.tile)-2,0,this.W),maxX=clamp(Math.ceil((this.camera.x+vw/2)/this.tile)+2,0,this.W),minY=clamp(Math.floor((this.camera.y-vh/2)/this.tile)-2,0,this.H),maxY=clamp(Math.ceil((this.camera.y+vh/2)/this.tile)+2,0,this.H);
    for(let y=minY;y<maxY;y++)for(let x=minX;x<maxX;x++)this.drawTile(x,y,this.tiles[this.idx(x,y)]);
    this.drawBuildings();
    if(this.overlayTerritory)this.drawGroups();
    if(this.overlayPath)this.drawPaths();
    this.drawCorpses();
    for(const a of this.agents)if(a.alive&&!a.turned)this.drawHuman(a);
    for(const z of this.zombies)if(z.alive)this.drawZombie(z);
    this.drawFires(minX,minY,maxX,maxY);
    if(this.overlayRad)this.drawRadiation(minX,minY,maxX,maxY);
    this.drawSelection();
    ctx.restore();
  }

  drawTile(x,y,t){
    const px=x*this.tile,py=y*this.tile; if(t.type==='grass'){ctx.fillStyle=t.variant%2?'#263427':'#29372a';ctx.fillRect(px,py,this.tile,this.tile);if(t.variant===3){ctx.fillStyle='#314231';ctx.fillRect(px+3,py+5,2,2)}}
    else if(t.type==='road'){ctx.fillStyle='#414441';ctx.fillRect(px,py,this.tile,this.tile);ctx.fillStyle='#4a4d49';if((x+y)%5===0)ctx.fillRect(px+2,py+2,5,1)}
    else if(t.type==='forest'){ctx.fillStyle='#1c2a20';ctx.fillRect(px,py,this.tile,this.tile);ctx.fillStyle='#263d2b';ctx.fillRect(px+3,py+2,10,9);ctx.fillStyle='#172119';ctx.fillRect(px+7,py+10,3,5)}
    else if(t.type==='water'){ctx.fillStyle='#1f3438';ctx.fillRect(px,py,this.tile,this.tile);ctx.fillStyle='#294248';ctx.fillRect(px+(t.variant*3)%12,py+6,7,1)}
    else {ctx.fillStyle='#353733';ctx.fillRect(px,py,this.tile,this.tile);}
  }

  drawBuildings(){

      const rad=this.radAt(b.x+b.w/2,b.y+b.h/2);ctx.fillStyle=b.type==='plant'?'#5a5750':b.type==='farm'?'#5a4f36':b.type==='hospital'?'#b6b4a5':b.type==='police'?'#66717a':b.type==='radio'?'#665c54':'#756d5d';ctx.fillRect(b.x+2,b.y+2,b.w-4,b.h-4);ctx.fillStyle='#252725';ctx.fillRect(b.x+5,b.y+b.h-8,b.w-10,6);
      if(!b.power){ctx.fillStyle='rgba(0,0,0,.35)';ctx.fillRect(b.x+2,b.y+2,b.w-4,b.h-4);} if(b.fort>.15){ctx.strokeStyle='#9b8b65';ctx.lineWidth=2;ctx.strokeRect(b.x+1,b.y+1,b.w-2,b.h-2);}
      if(this.camera.zoom>.7){ctx.font='7px monospace';ctx.textAlign='center';ctx.fillStyle='#d7d8cf';ctx.fillText(b.name.toUpperCase(),b.x+b.w/2,b.y-3);}

  drawCorpses(){
    for(const c of this.corpses){if(c.reanimated)continue;ctx.save();ctx.translate(Math.round(c.x),Math.round(c.y));ctx.fillStyle=c.infected?'#665f45':'#4d4a45';ctx.fillRect(-5,-1,10,3);ctx.fillStyle='#2b2c29';ctx.fillRect(-3,-3,4,2);ctx.restore();}
  }

  drawHuman(a){
    const r=4;ctx.save();ctx.translate(Math.round(a.x),Math.round(a.y));
    if(a.group){ctx.fillStyle=`hsla(${a.group.colorHue},45%,58%,.25)`;ctx.beginPath();ctx.arc(0,0,7,0,TAU);ctx.fill();}
    ctx.fillStyle=a.infection>.55?'#c9aa77':a.infection>.05?'#d0c39a':'#dad9d0';ctx.fillRect(-2,-5,4,4);ctx.fillStyle=a.stress>70?'#7b5e58':'#566676';ctx.fillRect(-3,-1,6,6);ctx.fillStyle='#242728';ctx.fillRect(-3,5,2,3);ctx.fillRect(1,5,2,3);
    if(a.weapon){ctx.fillStyle='#a59b7c';ctx.fillRect(4,0,4,1);} if(this.follow===a){ctx.strokeStyle='#eef4df';ctx.lineWidth=1;ctx.beginPath();ctx.arc(0,0,10,0,TAU);ctx.stroke();}ctx.restore();
  }

  drawZombie(z){ctx.save();ctx.translate(Math.round(z.x),Math.round(z.y));const glow=z.mutation>.45; if(glow){ctx.fillStyle='rgba(177,207,76,.16)';ctx.beginPath();ctx.arc(0,0,9+z.mutation*5,0,TAU);ctx.fill();}ctx.fillStyle=glow?'#a6c858':'#83906b';ctx.fillRect(-2,-5,4,4);ctx.fillStyle='#4c5948';ctx.fillRect(-3,-1,6,6);ctx.fillStyle='#2b302c';ctx.fillRect(-3,5,2,3);ctx.fillRect(1,5,2,3);if(this.follow===z){ctx.strokeStyle='#eef4df';ctx.beginPath();ctx.arc(0,0,10,0,TAU);ctx.stroke();}ctx.restore();}

  drawRadiation(minX,minY,maxX,maxY){ctx.save();for(let y=minY;y<maxY;y++)for(let x=minX;x<maxX;x++){const r=this.rad[this.idx(x,y)];if(r<.04)continue;ctx.fillStyle=`rgba(170,205,65,${Math.min(.46,r*.5)})`;ctx.fillRect(x*this.tile,y*this.tile,this.tile,this.tile);}ctx.restore();}
  drawFires(minX,minY,maxX,maxY){for(let y=minY;y<maxY;y++)for(let x=minX;x<maxX;x++){const f=this.fire[this.idx(x,y)];if(f<=0)continue;const px=x*this.tile+8,py=y*this.tile+8;ctx.fillStyle=`rgba(222,104,45,${.3+f*.45})`;ctx.beginPath();ctx.arc(px,py,4+f*5,0,TAU);ctx.fill();}}
  drawPaths(){ctx.save();ctx.lineWidth=1;for(const a of this.agents){if(a.path.length<2)continue;ctx.strokeStyle='rgba(196,210,205,.16)';ctx.beginPath();a.path.forEach((p,i)=>i?ctx.lineTo(p.x,p.y):ctx.moveTo(p.x,p.y));ctx.stroke();}ctx.restore();}
  drawGroups(){for(const g of this.groups){const members=[...g.members].map(id=>this.personById(id)).filter(a=>a&&a.alive&&!a.turned);if(members.length<2)continue;const cx=members.reduce((s,a)=>s+a.x,0)/members.length,cy=members.reduce((s,a)=>s+a.y,0)/members.length;ctx.fillStyle=`hsla(${g.colorHue},50%,55%,.08)`;ctx.beginPath();ctx.arc(cx,cy,40+members.length*5,0,TAU);ctx.fill();}}
  drawSelection(){const o=this.selected;if(!o)return;let x,y;if(o.kind==='building'){x=o.data.x+o.data.w/2;y=o.data.y+o.data.h/2;}else{x=o.x;y=o.y;}ctx.strokeStyle='#f0f2d6';ctx.lineWidth=1/this.camera.zoom;ctx.beginPath();ctx.arc(x,y,12,0,TAU);ctx.stroke();}
}
