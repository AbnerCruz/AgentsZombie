  saveSettings(key,light,deep,enabled){
    this.apiKey=(key||'').trim();this.lightModel=(light||'google/gemma-4-26b-a4b-it:free').trim();this.deepModel=(deep||'openai/gpt-oss-20b').trim();this.enabled=!!enabled;

    localStorage.setItem('az_light_model',this.lightModel);localStorage.setItem('az_deep_model',this.deepModel);localStorage.setItem('az_ai_enabled',this.enabled?'1':'0');
  }
  enqueue(agent,deep=false,trigger='routine'){
    if(!this.enabled||!this.apiKey||!agent.alive||agent.brain.pending)return false;
    agent.brain.pending=true;const job={agent,deep,trigger,enqueuedAt:performance.now()};if(deep)this.queue.unshift(job);else this.queue.push(job);return true;
  }
  tick(){
    if(this.inFlight>=this.maxConcurrent||!this.queue.length)return;
    const now=performance.now();if(now-this.lastDispatch<this.minDispatchMs)return;
    const job=this.queue.shift();if(!job.agent.alive){job.agent.brain.pending=false;return}
    this.lastDispatch=now;this.inFlight++;this.request(job).finally(()=>{this.inFlight--;job.agent.brain.pending=false});
  }
  memoryContext(a){
    const now=this.sim.minute;
    const relevant=a.memory.retrieve({now,tags:this.sim.currentMemoryTags(a),limit:14});
    return relevant.map(m=>({when:this.sim.formatMinute(m.minute),kind:m.kind,text:m.text,importance:m.importance}));
  }
  buildContext(a,trigger){
    const sensed=this.sim.sense(a);
    return {
      identity:{id:a.id,name:a.name,age:a.age,profession:a.profession,personality:a.personality,attributes:a.attr,personal_goal:a.brain.goal||null},
      time:this.sim.timeLabel(),trigger,
      body:{health:Math.round(a.health),hunger:Math.round(a.hunger),thirst:Math.round(a.thirst),energy:Math.round(a.energy),mood:Math.round(a.mood),stress:Math.round(a.stress),radiation:+a.radiation.toFixed(2),infection:+a.infection.toFixed(2),pain:Math.round(a.pain),encumbrance:+this.sim.inventoryWeight(a).toFixed(1)},
      inventory:a.inventory.map(i=>({id:i.id,name:i.name,qty:i.qty})),
      current_action:a.action?{tool:a.action.tool,args:a.action.args,progress:+(a.action.progress||0).toFixed(2)}:null,
      perceived:sensed,
      social:this.sim.socialContext(a),
      memories:this.memoryContext(a)
    };
  }
  async request(job){
    const a=job.agent,model=job.deep?this.deepModel:this.lightModel,context=this.buildContext(a,job.trigger);
    const sys=`Você é o cérebro individual de ${a.name}, uma pessoa vivendo o Dia 1 de um colapso zumbi e radioativo. Você NÃO é narrador, diretor nem controlador da simulação. Só conhece o que está no contexto sensorial, memórias e relações desta pessoa. Outros humanos podem ser agentes de IA ou NPCs, mas essa distinção não existe no mundo e você nunca deve inferi-la. Decida livremente o que esta pessoa tenta fazer usando UMA das ferramentas disponíveis. Não presuma resultados: a simulação resolve consequências. Preserve a personalidade e memórias. Pode agir de modo egoísta, altruísta, irracional, cooperativo ou cauteloso conforme esta pessoa. Não existe obrigação de formar grupo, construir, lutar ou sobreviver. Escolha uma ferramenta concreta.`;
    const body={model,messages:[{role:'system',content:sys},{role:'user',content:JSON.stringify(context)}],tools:AZ.AGENT_TOOLS,tool_choice:'required'};
    if(job.deep)body.reasoning={effort:'medium'};
    try{
      const res=await fetch('https://openrouter.ai/api/v1/chat/completions',{method:'POST',headers:{'Authorization':`Bearer ${this.apiKey}`,'Content-Type':'application/json','HTTP-Referer':location.origin+location.pathname,'X-Title':'Ashfall County'},body:JSON.stringify(body)});
      if(!res.ok)throw new Error(`OpenRouter ${res.status}: ${await res.text()}`);
      const data=await res.json();this.calls++;this.tokens+=(data.usage?.total_tokens||0);
      const msg=data.choices?.[0]?.message; const tc=msg?.tool_calls?.[0];
      if(!tc?.function?.name)throw new Error('Modelo não retornou ferramenta.');
      let args={};try{args=JSON.parse(tc.function.arguments||'{}')}catch{throw new Error('Argumentos de ferramenta inválidos.')}
      this.sim.executeAgentTool(a,tc.function.name,args,{model,deep:job.deep});
      a.brain.lastModel=model;a.brain.lastDecisionAt=this.sim.minute;a.brain.lastError='';
      if(msg.content)this.sim.remember(a,'thought',String(msg.content),4,['decision']);
    }catch(err){
      this.fails++;a.brain.lastError=String(err.message||err);a.brain.retryAfter=performance.now()+Math.min(30000,2500*Math.max(1,a.brain.failStreak||1));a.brain.failStreak=(a.brain.failStreak||0)+1;
      this.sim.remember(a,'system',`Não conseguiu concluir uma decisão externa: ${a.brain.lastError}`,2,['ai-error']);
    }
  }
