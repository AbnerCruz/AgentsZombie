  leadershipScore(a){
    const role=['policial','bombeiro','médico','enfermeira','professor','assistente social'].includes(a.job)?.12:0;
    return a.leadership*.42+a.courage*.25+a.empathy*.16+(1-a.stress/100)*.12+role;
  }

  electLeader(g){
    const members=[...g.members].map(id=>this.personById(id)).filter(a=>a&&a.alive&&!a.turned);
    if(!members.length){g.leaderId=null;return;}
    const leader=members.reduce((p,c)=>this.leadershipScore(c)>this.leadershipScore(p)?c:p);
    if(g.leaderId!==leader.id){
      const old=g.leaderId;g.leaderId=leader.id;
      if(old&&g.members.size>=4)this.log(this.timeText(),'LIDERANÇA',`${leader.name} passa a orientar ${g.name}.`);
    }
  }

  updateGroups(dt){
    for(const g of this.groups){
      const members=[...g.members].map(id=>this.personById(id)).filter(a=>a&&a.alive&&!a.turned);
      if(members.length<2)continue;
      if(!g.leaderId||!this.personById(g.leaderId)?.alive)this.electLeader(g);
      const avgStress=members.reduce((sum,a)=>sum+a.stress,0)/members.length;
      const avgTrust=members.reduce((sum,a)=>sum+this.relationship(a,this.personById(g.leaderId)||a),0)/members.length;
      g.cohesion=clamp(g.cohesion+(members.length*.00002+avgTrust*.00004-avgStress*.0000018)*dt,0,1);
      if(g.base&&members.length>=5&&g.cohesion>.46){
        for(const a of members)if(['wander','home'].includes(a.action)&&this.rng.chance(.06)){a.action='shelter';a.target={x:g.base.x+g.base.w/2,y:g.base.y+g.base.h/2};}
      }
      if(g.cohesion<.16&&members.length>3&&this.rng.chance(.008*dt)){
        const leaver=this.rng.pick(members.filter(a=>a.id!==g.leaderId));
        if(leaver){g.members.delete(leaver.id);leaver.group=null;this.remember(leaver,`Abandonou ${g.name} após conflitos.`);}
      }
    }
    for(const g of this.groups){if(g.members.size<2){for(const id of g.members){const m=this.personById(id);if(m&&m.group===g)m.group=null;}}}
    this.groups=this.groups.filter(g=>g.members.size>=2);
  }

  updateZombie(z,dt,minutes){
    z.decay+=minutes*.0005; z.rad=this.radAt(z.x,z.y);
    if(z.rad>.3)z.mutation=clamp(z.mutation+minutes*z.rad*.0009,0,1);
    let prey=this.nearestHuman(z,110+z.mutation*35);
    if(prey){z.target={x:prey.x,y:prey.y};}
    else{
      const pack=this.nearZombies(z,95).filter(o=>o!==z&&o.alive);
      const guided=pack.find(o=>o.target&&Math.hypot(o.target.x-o.x,o.target.y-o.y)>45);
      if(guided&&this.rng.chance(.16*dt)){z.target={x:guided.target.x+this.rng.range(-20,20),y:guided.target.y+this.rng.range(-20,20)};}
      else if(this.rng.chance(.015*dt))z.target=this.randomRoadPoint();
    }
    const dx=z.target.x-z.x,dy=z.target.y-z.y,d=Math.hypot(dx,dy)||1;
    const speed=(9+z.mutation*5)*(1-z.decay*.25);z.x=clamp(z.x+dx/d*speed*dt,5,this.W*this.tile-5);z.y=clamp(z.y+dy/d*speed*dt,5,this.H*this.tile-5);
    if(prey&&dist(z,prey)<13)this.zombieAttack(z,prey,dt);
  }

  }
