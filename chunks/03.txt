class RNG{constructor(seed){this.s=seed>>>0||1}next(){let x=this.s;x^=x<<13;x^=x>>>17;x^=x<<5;this.s=x>>>0;return this.s/4294967296}range(a,b){return a+(b-a)*this.next()}int(a,b){return Math.floor(this.range(a,b+1))}pick(a){return a[Math.floor(this.next()*a.length)]}chance(p){return this.next()<p}}

class MemoryStore{
 constructor(owner){this.owner=owner;this.entries=[];this.byTag=new Map();this.seq=1}
 add(minute,kind,text,importance=5,tags=[]){
   const m={id:this.seq++,minute,kind,text:String(text),importance:AZ.clamp(importance,1,10),tags:[...new Set(tags)]};this.entries.push(m);
   for(const t of m.tags){if(!this.byTag.has(t))this.byTag.set(t,[]);this.byTag.get(t).push(m)}return m;
 }
 retrieve({now,tags=[],limit=12}){
   const candidates=new Map();
   const recent=this.entries.slice(Math.max(0,this.entries.length-24));for(const m of recent)candidates.set(m.id,m);
   for(const t of tags)for(const m of (this.byTag.get(t)||[]))candidates.set(m.id,m);
   return [...candidates.values()].map(m=>({m,score:m.importance*3+Math.max(0,8-(now-m.minute)/180)+(m.tags.some(t=>tags.includes(t))?10:0)})).sort((a,b)=>b.score-a.score).slice(0,limit).map(x=>x.m).sort((a,b)=>a.minute-b.minute);
 }
}

class SpatialGrid{
 constructor(cell=220){this.cell=cell;this.map=new Map()}
 key(x,y){return `${Math.floor(x/this.cell)},${Math.floor(y/this.cell)}`}
 rebuild(items){this.map.clear();for(const e of items){if(!e.alive)continue;const k=this.key(e.x,e.y);if(!this.map.has(k))this.map.set(k,[]);this.map.get(k).push(e)}}
 near(x,y,r){const out=[],c=this.cell,cx=Math.floor(x/c),cy=Math.floor(y/c),n=Math.ceil(r/c),r2=r*r;for(let yy=cy-n;yy<=cy+n;yy++)for(let xx=cx-n;xx<=cx+n;xx++)for(const e of(this.map.get(`${xx},${yy}`)||[])){if((e.x-x)**2+(e.y-y)**2<=r2)out.push(e)}return out}
}
