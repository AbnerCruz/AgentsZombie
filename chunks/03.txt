  updateHuman(a,dt,minutes){

    if(a.hunger>82)a.health-=minutes*.018;if(a.thirst>78)a.health-=minutes*.032;
    if(a.infection>0){a.infection=clamp(a.infection+minutes*(.00062+.00034*(a.health<55)),0,1.2);a.health-=minutes*a.infection*.006;a.stress=clamp(a.stress+minutes*a.infection*.01,0,100);}
    if(a.infection>=1){this.turn(a);return;} if(a.health<=0){this.killHuman(a,'ferimentos e colapso físico');return;}

    // Perception.
    const dangerZ=this.nearestZombie(a,95);
    if(dangerZ){a.knownDanger=clamp(a.knownDanger+.25,0,1);a.stress=clamp(a.stress+minutes*.12,0,100); if(dist(a,dangerZ)<17){this.zombieAttack(dangerZ,a,dt);}}
    else a.knownDanger=Math.max(0,a.knownDanger-dt*.01);

    a.socialCooldown-=dt;

    this.moveHuman(a,dt,dangerZ);
    this.interactAtTarget(a,dt,minutes);
  }
