  createPopulation(agentCount,npcCount){
    const n=agentCount+npcCount;
    const homes=this.buildings.filter(b=>b.type==='home');
    const workplaces=this.buildings.filter(b=>['hospital','grocery','school','police','radio','fire','shop','pharmacy','farm','gas'].includes(b.type));
    const householdLast=[];
    for(let i=0;i<homes.length;i++) householdLast[i]=this.rng.pick(LAST);
    for(let i=0;i<n;i++){
      const hIndex=i%homes.length;
      const home=homes[hIndex];
      const age=this.rng.int(8,84);
      const job=age>67?'aposentado':age<18?'estudante':this.rng.pick(JOBS);
      const work=this.findWorkplaceForJob(job,workplaces);
      const cognitive=i<agentCount?'agent':'npc'; // Internal simulation detail; never exposed to inhabitants.
      const a={
        kind:'human',id:i+1,name:`${this.rng.pick(FIRST)} ${householdLast[hIndex]}`,age,job,trait:this.rng.pick(TRAITS),cognitive,
        x:home.x+this.rng.range(8,home.w-8),y:home.y+this.rng.range(8,home.h-8),vx:0,vy:0,
        home,work,health:100,hunger:this.rng.range(3,20),thirst:this.rng.range(2,18),fatigue:this.rng.range(2,22),stress:this.rng.range(4,18),radiation:0,infection:0,
        courage:this.rng.range(.25,.85),empathy:this.rng.range(.25,.9),sociability:this.rng.range(.2,.95),awareness:this.rng.range(.3,.85),
        leadership:this.rng.range(.15,.92),risk:this.rng.range(.15,.9),
        food:this.rng.int(0,2),water:this.rng.int(0,2),meds:0,weapon:this.rng.chance(.12),hasRadio:this.rng.chance(.38),alive:true,turned:false,
        action:'home',target:{x:home.x+home.w/2,y:home.y+home.h/2},group:null,memories:[],path:[],decisionIn:this.rng.range(0,.8),panic:false,knownDanger:0,radioKnowledge:0,lastBuilding:home,
        family:[],kills:0,helped:0,beliefs:new Set(),rumors:[],relations:new Map(),socialCooldown:this.rng.range(2,12),goal:null
      };
      this.agents.push(a);
    }
    // Household relationships are real to everyone; nobody has access to simulation tiers.
    for(const a of this.agents){
      a.family=this.agents.filter(b=>b!==a&&b.home===a.home).map(b=>b.id);
      for(const id of a.family)a.relations.set(id,this.rng.range(.58,.92));
    }
    this.seedMorningRoutine();
  }

  seedMorningRoutine(){
    for(const a of this.agents){
      const roll=this.rng.next();
      if(a.work && roll<.38){
        a.x=a.work.x+this.rng.range(8,Math.max(9,a.work.w-8));
        a.y=a.work.y+this.rng.range(8,Math.max(9,a.work.h-8));
        a.lastBuilding=a.work;
        a.action='wander';
      }else if(a.work && roll<.72){
        const t=this.rng.range(.15,.85);
        a.x=lerp(a.home.x+a.home.w/2,a.work.x+a.work.w/2,t)+this.rng.range(-20,20);
        a.y=lerp(a.home.y+a.home.h/2,a.work.y+a.work.h/2,t)+this.rng.range(-20,20);
        a.target={x:a.work.x+a.work.w/2,y:a.work.y+a.work.h/2};
        a.action='wander';
      }
      if(a.job==='policial'||a.job==='bombeiro'||a.job==='enfermeira'||a.job==='médico')a.beliefs.add('emergency-traffic');
      if(a.job==='radialista')a.beliefs.add('radio-active');
      if(a.job==='técnica de laboratório'&&this.rng.chance(.55))a.beliefs.add('cinder-anomaly');
    }
  }

  findWorkplaceForJob(job, workplaces){
    const map={enfermeira:'hospital',médico:'hospital',professor:'school',policial:'police',bombeiro:'fire',radialista:'radio',mecânico:'shop',caixa:'grocery',farmacêutica:'pharmacy',fazendeiro:'farm',motorista:'gas',cozinheira:'grocery',segurança:'police','técnica de laboratório':'hospital',carpinteiro:'shop',estudante:'school'};
    const t=map[job];
    return workplaces.find(b=>b.type===t) || this.rng.pick(workplaces);
  }
