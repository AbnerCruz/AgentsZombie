 executeAgentTool(a,name,args,meta={}){
   if(!a.alive)return;const reject=why=>{this.remember(a,'decision',`Tentou ${name}, mas não pôde: ${why}`,4,['decision']);a.action=null;a.brain.nextDecision=this.minute+3};
   const setAction=(tool,data,dur=0)=>{a.action={tool,args:data,started:this.minute,duration:dur,progress:0,model:meta.model||'local'};a.brain.nextDecision=this.minute+Math.max(4,dur*.7);this.remember(a,'decision',`${name}: ${data.reason||''}`.trim(),5,['decision',name])};
   if(name==='move'){
     const d=DIRECTIONS[args.direction];if(!d)return reject('direção inválida');const distance=AZ.clamp(+args.distance||100,20,900);a.target={x:a.x+d[0]*distance,y:a.y+d[1]*distance};return setAction(name,args,distance/45);
   }
   if(name==='go_to'){
     const t=this.entityById(args.target_id);if(!t)return reject('alvo desconhecido');a.target={x:t.x+(t.w||0)/2,y:t.y+(t.h||0)/2,id:t.id};return setAction(name,args,Math.hypot(a.target.x-a.x,a.target.y-a.y)/45);
   }
   if(name==='flee'){
     const d=DIRECTIONS[args.direction];if(!d)return reject('direção inválida');a.target={x:a.x+d[0]*520,y:a.y+d[1]*520};a.stress=Math.min(100,a.stress+4);return setAction(name,args,10);
   }
   if(name==='consume'){
     if(!this.itemQty(a,args.item_id))return reject('item ausente');this.takeItem(a,args.item_id,1);
     if(args.item_id==='water')a.thirst=Math.max(0,a.thirst-46);else if(args.item_id==='food')a.hunger=Math.max(0,a.hunger-38);else if(args.item_id==='bandage'){a.health=Math.min(100,a.health+12);a.pain=Math.max(0,a.pain-18)}else if(args.item_id==='antirad')a.radiation=Math.max(0,a.radiation-.32);
