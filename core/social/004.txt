function settleTrade(a,b,trade){if(!trade||trade.status!=='accept')return false;const oq=Math.max(0,trade.offer_qty|0),rq=Math.max(0,trade.request_qty|0),oi=trade.offer_item,ri=trade.request_item;if(oq&&sim.itemQty(a,oi)<oq)return false;if(rq&&sim.itemQty(b,ri)<rq)return false;const given=localValue(a,oi,oq),received=localValue(a,ri,rq);if(oq&&!transfer(a,b,oi,oq))return false;if(rq&&!transfer(b,a,ri,rq)){if(oq)transfer(b,a,oi,oq);return false}const ab=axis(a,b),ba=axis(b,a);ab.trust=clamp(ab.trust+4,-100,100);ba.trust=clamp(ba.trust+4,-100,100);ab.respect=clamp(ab.respect+2,-100,100);ba.respect=clamp(ba.respect+2,-100,100);if(trade.credit){const diff=Math.round(received-given);if(diff>0){ab.debt=clamp(ab.debt+diff,-100,100);ba.debt=clamp(ba.debt-diff,-100,100);addRepayPromise(a,b,diff)}else if(diff<0){const v=Math.abs(diff);ba.debt=clamp(ba.debt+v,-100,100);ab.debt=clamp(ab.debt-v,-100,100);addRepayPromise(b,a,v)}}sim.practice(a,'social',2);sim.practice(b,'social',1);sim.remember(a,'trade',`Concluiu uma troca com ${b.name}.`,6,['social','trade',b.id]);sim.remember(b,'trade',`Concluiu uma troca com ${a.name}.`,6,['social','trade',a.id]);return true}
