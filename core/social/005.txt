async function runScene(job){const parts=job.parts.filter(p=>p?.alive);if(parts.length<2)return false;const key=job.key;if(S.pending.get(key)!==job)return false;if(!ai.enabled||!ai.apiKey){S.pending.delete(key);return false}if(ai.activeBrains.size>=ai.maxConcurrent)return false;const token=`scene:${key}:${Date.now()}`;ai.activeBrains.add(token);try{const ctx=sceneContext(parts,job.seed),model=ai.socialModel||ai.lightModel,sys='Resolva uma cena social realista entre sobreviventes. Você controla apenas fala, interpretação social, informação transmitida, promessas, mentira/omissão e negociação. Não mova corpos, não invente itens, não resolva combate. Cada participante só pode usar o que aparece no próprio contexto. Produza 3 a 5 turnos quando fizer sentido. Dizer algo não significa acreditar nisso. Relações têm eixos independentes. Retorne somente o JSON do schema.';const data=await ai.fetchCompletion({model,messages:[{role:'system',content:sys},{role:'user',content:JSON.stringify(ctx)}],response_format:{type:'json_schema',json_schema:schema()}},token,26000);ai.calls++;ai.socialCalls=(ai.socialCalls||0)+1;ai.accountUsage(data.usage,model,parts[0]);const out=parseJSON(data.choices?.[0]?.message?.content||'');if(!out)throw new Error('Cena social sem JSON válido');applyRelations(parts,out);applyBeliefs(parts,out);applyTheory(parts,out);applyPromises(parts,out);rememberScene(parts,out);if(job.mode==='trade'){const t=out.outcome?.trade;if(t?.status==='accept')settleTrade(job.a,job.b,t);else{sim.remember(job.a,'trade',t?.status==='counter'?`${job.b.name} fez uma contraproposta.`:`${job.b.name} recusou a troca.`,5,['social','trade',job.b.id]);if(t?.status==='counter')V.markReplan?.(job.a,'trade_counteroffer',false)}}S.scenes.push({id:`sc:${++S.sceneSeq}`,minute:sim.minute,participants:parts.map(p=>p.id),turns:out.turns||[],outcome:out.outcome||{},mode:job.mode,situation:job.situation});if(S.scenes.length>180)S.scenes.splice(0,S.scenes.length-180);S.pending.delete(key);return true}catch(e){ai.fails++;ai.lastError=String(e?.message||e);S.pending.delete(key);return false}finally{ai.activeBrains.delete(token)}}
function collect(a,b){const out=[a,b],site=sim.currentSite?.(a);for(const p of sim.peopleGrid.near(a.x,a.y,90)){if(out.length>=5)break;if(!p.alive||out.includes(p))continue;if(site&&sim.currentSite?.(p)?.id!==site.id)continue;if(Math.hypot(p.x-a.x,p.y-a.y)<75)out.push(p)}return out}
S.requestScene=function(a,b,seed={},mode='talk'){ensure(a);ensure(b);const k=pairKey(a,b),last=S.scenes.slice().reverse().find(x=>x.participants.includes(a.id)&&x.participants.includes(b.id));const situation=JSON.stringify([mode,seed.intent||'',seed.message||'',Math.floor(sim.minute/6),sim.currentSite?.(a)?.id||'out']);if(last&&last.situation===situation)return false;const job={key:k,parts:collect(a,b),a,b,seed,mode,situation,created:performance.now()};S.pending.set(k,job);runScene(job);return true};
