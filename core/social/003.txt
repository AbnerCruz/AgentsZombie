function applyTheory(parts,out){for(const t of out.theory_updates||[]){const a=parts.find(x=>x.id===t.observer_id),b=parts.find(x=>x.id===t.target_id);if(!a||!b||a===b)continue;const s=axis(a,b);s.estimatedGoal=String(t.estimated_goal||'').slice(0,300)||null;for(const k of t.known_fact_keys||[])s.estimatedKnowledge[k]=true}for(const v of out.vocabulary_updates||[]){const p=parts.find(x=>x.id===v.participant_id);if(!p)continue;p.lexicon=p.lexicon||{};p.lexicon[String(v.concept||'concept')]=String(v.label||'').slice(0,80)}}
function applyPromises(parts,out){for(const pr of out.promises||[]){const from=parts.find(x=>x.id===pr.from_id),to=parts.find(x=>x.id===pr.to_id);if(!from||!to)continue;const obj={id:`pr:${++S.sceneSeq}`,from:from.id,to:to.id,type:pr.type,content:pr.content,created:sim.minute,due:sim.minute+(pr.type==='meet'?240:pr.type==='repay'?720:360),status:'pending'};from.promises.push(obj);to.promises.push(obj);axis(from,to).promises.push(obj.id);axis(to,from).promises.push(obj.id)}}
function itemTemplate(p,id){const i=(p.inventory||[]).find(x=>x.id===id);return i?{...i,qty:1}:null}
function transfer(src,dst,id,qty){qty=Math.max(0,qty|0);if(!qty)return true;const item=itemTemplate(src,id);if(!item||sim.itemQty(src,id)<qty)return false;if(!sim.takeItem(src,id,qty))return false;const before=(dst.inventory||[]).find(x=>x.id===id);sim.addItem(dst,item,qty);const after=(dst.inventory||[]).find(x=>x.id===id);if(after){const keepQty=after.qty;Object.assign(after,item);after.qty=keepQty}return true}
S.transferItem=transfer;
function localValue(p,id,qty=1){const item=(p.inventory||[]).find(x=>x.id===id)||{id,kind:'misc',weight:.5},need=(id==='water'?p.thirst:id==='food'?p.hunger:item.kind==='medical'?(100-p.health):item.kind==='ammo'?35:20),scarcity=1/(1+sim.itemQty(p,id));const kind={weapon:18,medical:16,drink:14,food:12,ammo:10,fuel:10,tool:9,material:5,misc:3}[item.kind]||4;return Math.max(1,qty*(kind+need*.22)*(1+scarcity*.7))}
function addRepayPromise(debtor,creditor,value){if(value<4)return;const obj={id:`pr:${++S.sceneSeq}`,from:debtor.id,to:creditor.id,type:'repay',content:`Repor valor aproximado ${Math.round(value)} em recursos ou ajuda.`,created:sim.minute,due:sim.minute+720,status:'pending'};debtor.promises.push(obj);creditor.promises.push(obj);axis(debtor,creditor).promises.push(obj.id);axis(creditor,debtor).promises.push(obj.id)}
