function inferGroups(){const ppl=sim.people.filter(p=>p.alive&&!p.military),adj=new Map(ppl.map(p=>[p.id,[]]));for(let i=0;i<ppl.length;i++)for(let j=i+1;j<ppl.length;j++){const a=ppl[i],b=ppl[j],ab=axis(a,b),ba=axis(b,a),link=Math.min(ab.familiarity,ba.familiarity)>24&&((ab.trust+ba.trust)/2>8||Math.min(ab.sharedTime,ba.sharedTime)>35);if(link){adj.get(a.id).push(b.id);adj.get(b.id).push(a.id)}}const seen=new Set(),groups=[];for(const p of ppl){if(seen.has(p.id))continue;const stack=[p.id],ids=[];seen.add(p.id);while(stack.length){const id=stack.pop();ids.push(id);for(const q of adj.get(id)||[])if(!seen.has(q)){seen.add(q);stack.push(q)}}if(ids.length<2)continue;const members=ids.map(id=>sim.entityById(id)).filter(Boolean),leader=members.slice().sort((a,b)=>{const score=x=>members.filter(q=>q!==x).reduce((n,q)=>n+(axis(q,x).respect||0)+(axis(q,x).trust||0)*.35,0)+sim.skillLevel(x,'social')*5;return score(b)-score(a)})[0],scarce=members.reduce((n,x)=>n+x.hunger+x.thirst,0)/(members.length*2),norms=[];if(scarce>62)norms.push('racionar recursos');if(members.some(x=>x.wounds?.length))norms.push('priorizar feridos');if(members.some(x=>x.knowledge?.zombieSeen))norms.push('manter vigília');groups.push({id:`g:${ids.slice().sort().join('.')}`,members:ids,leader:leader?.id||null,norms,minute:sim.minute})}const before=new Map((S.groups||[]).map(g=>[g.id,g]));for(const g of groups)if(!before.has(g.id))S.groupHistory.push({minute:sim.minute,type:'formed',group:g});for(const g of S.groups||[])if(!groups.some(x=>x.id===g.id))S.groupHistory.push({minute:sim.minute,type:'split_or_dissolved',group:g});if(S.groupHistory.length>120)S.groupHistory.splice(0,S.groupHistory.length-120);S.groups=groups}
setInterval(()=>{for(const job of S.pending.values())if(performance.now()-job.created<45000)runScene(job);else S.pending.delete(job.key);socialObservation();promiseTick();inferGroups()},1400);
S.ready=true;
})();
