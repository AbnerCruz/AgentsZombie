// Spatial hash incremental: preserva a API near()/rebuild(), mas deixa de destruir e recriar buckets todo frame.
class IncrementalGrid{
  constructor(cell=220){this.cell=cell;this.map=new Map();this.keys=new WeakMap();this.members=new Set()}
  key(x,y){return`${Math.floor(x/this.cell)},${Math.floor(y/this.cell)}`}
  _add(e,k){let b=this.map.get(k);if(!b)this.map.set(k,b=new Set());b.add(e);this.keys.set(e,k);this.members.add(e)}
  _remove(e){const k=this.keys.get(e);if(k!=null){const b=this.map.get(k);if(b){b.delete(e);if(!b.size)this.map.delete(k)}}this.keys.delete(e);this.members.delete(e)}
  sync(items){
    for(const e of items){
      if(!e?.alive){if(this.members.has(e))this._remove(e);continue}
      const k=this.key(e.x,e.y),old=this.keys.get(e);
      if(old===k)continue;
      if(old!=null)this._remove(e);
      this._add(e,k);perf.gridMoves++;
    }
  }
  rebuild(items){this.sync(items)}
  near(x,y,r){const out=[],c=this.cell,cx=Math.floor(x/c),cy=Math.floor(y/c),n=Math.ceil(r/c),r2=r*r;for(let yy=cy-n;yy<=cy+n;yy++)for(let xx=cx-n;xx<=cx+n;xx++){const b=this.map.get(`${xx},${yy}`);if(!b)continue;for(const e of b)if(e.alive&&(e.x-x)**2+(e.y-y)**2<=r2)out.push(e)}return out}
}
