function namesFor(trigger,a){const t=String(trigger||'').toLowerCase();let names;if(/threat|wound|attacked|death|combat|grab|danger/.test(t))names=['flee','attack','move','go_to','aid','consume','combat_stance','talk'];else if(/social|speech|conversation|promise|trade|group/.test(t))names=['talk','trade','aid','go_to','set_goal','wait'];else if(/resource|hunger|thirst|body|sleep|scavenge|material/.test(t))names=['consume','scavenge','go_to','move','rest','craft','enter_vehicle','drive_vehicle','refuel_vehicle'];else if(/barrier|blocked|path|door|construction/.test(t))names=['go_to','move','build','scavenge','craft','flee'];else if(/vehicle|fuel/.test(t))names=['enter_vehicle','drive_vehicle','exit_vehicle','refuel_vehicle','go_to','move'];else names=['move','go_to','scavenge','consume','rest','talk','craft','enter_vehicle','drive_vehicle','set_goal'];if(a?.vehicleId)names.push('exit_vehicle');return[...new Set(names)].map(n=>toolMap.get(n)).filter(Boolean)}
ai.selectTools=function(trigger,a){return namesFor(trigger,a)};
